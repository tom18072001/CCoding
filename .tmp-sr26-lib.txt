
// ---------- SR-26: p hoán vị hai mẫu (H3) và p bootstrap có quy tắc trần/suy biến (H6) ----------

/** Cổng nhãn: chỉ "xác nhận" khi p nhỏ nhất phép kiểm có thể cho ≤ ngưỡng Holm. */
export function confirmable(pMin, holmThreshold) {
  return Number.isFinite(pMin) && Number.isFinite(holmThreshold) && pMin <= holmThreshold;
}

const choose = (n, k) => {
  let r = 1;
  for (let i = 1; i <= k; i++) r = (r * (n - k + i)) / i;
  return Math.round(r);
};

/**
 * Hoán vị hai mẫu chính xác: liệt kê mọi cách chọn b.length lượt làm TREAT, N = C(ra + rb, rb).
 * Điểm TREAT được dịch trước: b′ = b + shift (recall H3: shift = −0,15; precision H3: shift = +0,05).
 * T = mean(b′) − mean(a); alt 'greater' | 'less' | 'two'; dung sai 1e-12.
 */
export function permTwoSample(a, b, { shift = 0, alt = 'greater' } = {}) {
  const ra = a.length, rb = b.length;
  if (!ra || !rb) return { p: null, pMin: null, N: 0, exact: true };
  const bp = b.map((x) => x + shift);
  const pool = [...a, ...bp], n = pool.length, tot = pool.reduce((s, v) => s + v, 0);
  const T = (sel) => sel / rb - (tot - sel) / ra;
  const tObs = T(bp.reduce((s, v) => s + v, 0));
  const N = choose(n, rb);
  if (N > 5e7) throw new Error(`permTwoSample: N = ${N} quá lớn để liệt kê`);
  let k = 0;
  const hit = (t) => (alt === 'less' ? t <= tObs + EPS : alt === 'two' ? Math.abs(t) >= Math.abs(tObs) - EPS : t >= tObs - EPS);
  const rec = (start, left, sum) => {
    if (left === 0) { if (hit(T(sum))) k++; return; }
    for (let i = start; i <= n - left; i++) rec(i + 1, left - 1, sum + pool[i]);
  };
  rec(0, rb, 0);
  return { p: k / N, pMin: 1 / N, N, exact: true };
}

/** Tập bài H6 khai báo trước (preregistration §2.5 (a)). */
export const H6_TASKS = ['l1-modules', 'l2-chain', 't1', 't2', 't3', 't4', 't5'];

const H6_MARGIN = 0.05;
const NO_CONCLUSION = 'Không kết luận (trần/suy biến)';

/**
 * Bootstrap p cho H6, MỘT cặp (a = B0, b = A1 hoặc A2; `arm` 1|2 chỉ để đặt nA1/nA2).
 * (a) chỉ taskSet; (b) loại bài mà mọi lượt cả hai arm ≥ ceil, giữ bài chạm trần một phía;
 * (c) Δ* suy biến (mọi Δ* bằng nhau, dung sai 1e-12) ⇒ p = null, không bao giờ 1/(B+1);
 * (d) < minTasks bài ⇒ verdict 'estimate' kèm ci90, không p.
 * p_dưới = (k+1)/(B+1) với k = #{Δ* ≤ −0,05}; p_trên với #{Δ* ≥ +0,05}; p = max.
 */
export function bootP(cells, { B = 4000, seed = 20261102, ceil = 0.99, minTasks = 3, taskSet = H6_TASKS, arm = 1 } = {}) {
  const inSet = new Set(taskSet);
  const ceilingTasks = [], kept = [];
  for (const c of cells) {
    if (!inSet.has(c.task) || !c.a?.length || !c.b?.length) continue;
    if ([...c.a, ...c.b].every((x) => x >= ceil)) ceilingTasks.push(c.task);
    else kept.push(c);
  }
  const n = kept.length;
  const base = { ceilingTasks, B, k: null, nA1: arm === 1 ? n : null, nA2: arm === 2 ? n : null };
  const rnd = mulberry32(seed);
  const res = (x) => x.map(() => x[Math.floor(rnd() * x.length)]);
  const deltas = [];
  if (n) {
    for (let i = 0; i < B; i++) deltas.push(mean(kept.map((c) => mean(res(c.b)) - mean(res(c.a)))));
  }
  const sorted = [...deltas].sort((x, y) => x - y);
  const ci90 = sorted.length ? [quantile(sorted, 0.05), quantile(sorted, 0.95)] : null;
  if (n < minTasks) return { ...base, p: null, verdict: 'estimate', ci90 };
  if (sorted[sorted.length - 1] - sorted[0] <= EPS) {
    const outside = deltas.every((d) => Math.abs(d) >= H6_MARGIN - EPS);
    return { ...base, p: null, verdict: outside ? 'Bác bỏ' : NO_CONCLUSION, ci90 };
  }
  let kLo = 0, kHi = 0;
  for (const d of deltas) { if (d <= -H6_MARGIN + EPS) kLo++; if (d >= H6_MARGIN - EPS) kHi++; }
  const k = Math.max(kLo, kHi);
  return { ...base, p: (k + 1) / (B + 1), pMin: 1 / (B + 1), verdict: 'p', ci90, k };
}

/** Gộp hai cặp: một cặp NA ⇒ p(H6) = null (không max); chỉ khi cả hai có p mới lấy max. */
export function h6Combine(pA1, pA2) {
  const out = { nA1: pA1.nA1, nA2: pA2.nA2, ceilingTasks: { A1: pA1.ceilingTasks, A2: pA2.ceilingTasks }, ci90: { A1: pA1.ci90, A2: pA2.ci90 }, B: pA1.B };
  const vs = [pA1.verdict, pA2.verdict];
  if (vs.includes('estimate')) return { ...out, p: null, verdict: 'estimate', k: null };
  if (pA1.p == null || pA2.p == null) {
    return { ...out, p: null, verdict: vs.includes('Bác bỏ') ? 'Bác bỏ' : NO_CONCLUSION, k: null };
  }
  const worse = pA1.p >= pA2.p ? pA1 : pA2;
  return { ...out, p: worse.p, pMin: Math.max(pA1.pMin, pA2.pMin), verdict: 'p', k: worse.k };
}
